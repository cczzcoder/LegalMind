from functools import lru_cache
from typing import Literal
from uuid import UUID

from pydantic import SecretStr, model_validator
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    model_config = SettingsConfigDict(
        env_file=".env",
        extra="ignore",
    )

    app_env: Literal["development", "test", "production"] = "development"
    database_url: str
    dev_api_key: SecretStr
    dev_org_id: UUID
    dev_user_id: UUID

    @model_validator(mode="after")
    def validate_security_mode(self):
        if self.app_env == "production":
            raise ValueError(
                "Production is disabled until real authentication "
                "and authorization are implemented."
            )

        key = self.dev_api_key.get_secret_value()
        if len(key) < 32 or key.startswith("REPLACE"):
            raise ValueError("Configure a random DEV_API_KEY.")

        return self


@lru_cache
def get_settings() -> Settings:
    return Settings()

import secrets
from dataclasses import dataclass
from typing import Annotated
from uuid import UUID

from fastapi import Depends, HTTPException
from fastapi.security import APIKeyHeader

from app.core.config import get_settings


api_key_header = APIKeyHeader(
    name="X-Dev-Key",
    auto_error=False,
)


@dataclass(frozen=True)
class Principal:
    organization_id: UUID
    user_id: UUID


async def get_principal(
    key: Annotated[str | None, Depends(api_key_header)],
) -> Principal:
    settings = get_settings()
    expected = settings.dev_api_key.get_secret_value()

    if key is None or not secrets.compare_digest(
        key.encode("utf-8"),
        expected.encode("utf-8"),
    ):
        raise HTTPException(
            status_code=401,
            detail="Invalid development credentials",
        )

    return Principal(
        organization_id=settings.dev_org_id,
        user_id=settings.dev_user_id,
    )

from uuid import UUID

from fastapi import HTTPException
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.security import Principal
from app.models import (
    AuditEvent,
    OutboxEvent,
    WikiPage,
    WikiRevision,
)
from app.modules.wiki.schemas import CreatePage, CreateRevision


def record_change(
    session: AsyncSession,
    principal: Principal,
    page_id: UUID,
    revision: WikiRevision,
    action: str,
) -> None:
    payload = {
        "page_id": str(page_id),
        "revision_id": str(revision.id),
        "revision_number": revision.number,
    }

    session.add(
        AuditEvent(
            organization_id=principal.organization_id,
            actor_id=principal.user_id,
            action=action,
            resource_id=page_id,
            payload=payload,
        )
    )

    session.add(
        OutboxEvent(
            organization_id=principal.organization_id,
            event_type=action,
            payload=payload,
        )
    )


async def create_page(
    session: AsyncSession,
    principal: Principal,
    data: CreatePage,
) -> WikiRevision:
    async with session.begin():
        page = WikiPage(
            organization_id=principal.organization_id,
            title=data.title,
            head_revision=1,
        )
        session.add(page)
        await session.flush()

        revision = WikiRevision(
            page_id=page.id,
            number=1,
            body=data.body,
            author_id=principal.user_id,
        )
        session.add(revision)
        await session.flush()

        record_change(
            session,
            principal,
            page.id,
            revision,
            "wiki.page.created",
        )

        await session.flush()
        await session.refresh(revision)

    return revision


async def create_revision(
    session: AsyncSession,
    principal: Principal,
    page_id: UUID,
    data: CreateRevision,
) -> WikiRevision:
    async with session.begin():
        page = await session.scalar(
            select(WikiPage)
            .where(
                WikiPage.id == page_id,
                WikiPage.organization_id == principal.organization_id,
            )
            .with_for_update()
        )

        if page is None:
            raise HTTPException(
                status_code=404,
                detail="Page not found",
            )

        if page.head_revision != data.expected_revision:
            raise HTTPException(
                status_code=409,
                detail="Revision conflict: reload before editing",
            )

        page.head_revision += 1

        revision = WikiRevision(
            page_id=page.id,
            number=page.head_revision,
            body=data.body,
            author_id=principal.user_id,
        )
        session.add(revision)
        await session.flush()

        record_change(
            session,
            principal,
            page.id,
            revision,
            "wiki.revision.created",
        )

        await session.flush()
        await session.refresh(revision)

    return revision
''',

import {
  Alert,

} from "antd";

type Page = {
  id: string;
  title: string;
  head_revision: number;
  created_at: string;
};

type Revision = {
  id: string;
  page_id: string;
  number: number;
  body: string;
  status: string;
  author_id: string;
  created_at: string;
};

export default function App() {
  const [key, setKey] = useState("");
  const [pages, setPages] = useState<Page[]>([]);
  const [selected, setSelected] = useState<Page | null>(null);
  const [revisions, setRevisions] = useState<Revision[]>([]);
  const [title, setTitle] = useState("");
  const [body, setBody] = useState("");
  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");
  const [busy, setBusy] = useState(false);

  async function api<T>(
    path: string,
    init: RequestInit = {},
  ): Promise<T> {
    const headers = new Headers(init.headers);
    headers.set("X-Dev-Key", key);

    if (init.body !== undefined) {
      headers.set("Content-Type", "application/json");
    }

    const response = await fetch(`/api/v1${path}`, {
      ...init,
      headers,
    });

    if (!response.ok) {
      const payload = await response.json().catch(() => null);
      throw new Error(
        `${response.status}: ${
          payload?.detail
            ? JSON.stringify(payload.detail)
            : response.statusText
        }`,
      );
    }

    return response.json() as Promise<T>;
  }

  async function execute(task: () => Promise<void>) {
    setBusy(true);
    setError("");
    setNotice("");

    try {
      await task();
    } catch (reason) {
      setError(
        reason instanceof Error ? reason.message : String(reason),
      );
    } finally {
      setBusy(false);
    }
  }

  async function loadPages() {
    const result = await api<Page[]>("/wiki/pages?limit=100");
    setPages(result);
  }

  async function openPage(page: Page) {
    const result = await api<Revision[]>(
      `/wiki/pages/${page.id}/revisions?limit=100`,
    );

    if (!result.length) {
      throw new Error("该页面没有可读取修订。");
    }

    setSelected({
      ...page,
      head_revision: result[0].number,
    });
    setTitle(page.title);
    setBody(result[0].body);
    setRevisions(result);
  }

  function newPage() {
    setSelected(null);
    setRevisions([]);
    setTitle("");
    setBody("");
    setError("");
    setNotice("");
  }

  async function saveDraft() {
    if (selected) {
      const revision = await api<Revision>(
        `/wiki/pages/${selected.id}/revisions`,
        {
          method: "POST",
          body: JSON.stringify({
            expected_revision: selected.head_revision,
            body,
          }),
        },
      );

      setSelected({
        ...selected,
        head_revision: revision.number,
      });
      setRevisions((previous) => [revision, ...previous]);
      setNotice(`已保存为修订 ${revision.number}。`);
    } else {
      const revision = await api<Revision>("/wiki/pages", {

   - 迁移账号与业务账号分离。
   - 当前 Compose 使用的是开发账号，不是生产最小权限配置。

3. **锁定依赖**
   - Python 生成并提交锁文件。
   - 前端提交 `package-lock.json`，构建改为 `npm ci`。
   - 锁定经过测试的镜像摘要。
   - 编排框架（LangGraph、LangChain）及其传递依赖同样纳入锁定，并核查许可证与已知漏洞。

4. **替换前端开发服务器**
   - 正式环境使用构建产物及反向代理。
   - 配置 HTTPS、请求大小限制和超时。

5. **完善审计**
   - 当前只是事务内记录。
   - 尚无数据库级 append-only 权限、归档和外部校验点。

6. **完善任务可靠性**
   - Outbox 表不等于可靠任务系统。
   - 需要投递器、幂等消费、重试、超时和对账。

7. **补集成测试和恢复演练**
   - 尤其是跨组织访问、事务回滚、重复任务、数据库恢复。

8. **评估图谱存储**
   - GraphRAG 概念验证阶段基于 PostgreSQL + pgvector，不新增常驻组件。
   - 迁移 Neo4j 前，须完成准确率评测并满足设计文档第 18 节触发条件，同时补充图存储的备份、恢复和权限隔离方案。

9. **真实资料入库前先做权限与来源**
   - 不要先把客户资料导进去，再补权限与外发控制。

10. **合规前置检查**（设计文档第 20 节、需求 FR-14）
   - 确认系统定位为法律信息辅助工具；页面、导出物与问答输出均带“不构成法律意见”声明与知识范围说明。
   - 确认每个来源均已登记授权说明（`license_note`）与最后核查时间；官方渠道优先，非官方来源有授权依据。
   - **逐份核对原件的来源归属**，不能只看文件标题——同一批文件可能来自不同来源（官方库与第三方数据库并存）。归属登记有误时用 `PUT /api/v1/documents/{id}/source` 或 `app.cli set-document-source`（需 `source.manage`）更正；来源登记信息本身有误时用 `PUT /api/v1/sources/{id}` 或 `app.cli update-source`。两者都写审计（含变更前后值），因为设计文档 §20.3 要求「来源更新、撤回或内容变化须留痕」。
   - ⚠️ **版权声明不是授权说明**。第三方数据库页脚常见的「版权所有：某公司 + ICP 备案号」只表明该库主张权利，**不构成我们存储与使用的授权**。授权说明必须写明本机构的使用依据（官方公开、订阅协议、书面许可等）与使用范围；**未取得授权的来源，其内容不得入库**（§20.3）。**不得为了通过检查而填写与事实不符的授权说明**——审计会记录每次变更前后的原文。
   - **来源被认定无权使用时须撤下其原件**：`python -m app.cli withdraw-document --as kadmin --from-source <来源 ID> --reason "原因"`（先加 `--dry-run` 预演）。撤下会删除原件、解析版本、分块、定位与脱敏映射（设计文档 §15.3 要求删除覆盖原件与解析产物），**不可逆**。同一版本若还有别的合法来源原件，版本树会从它重建，不会被连带删除。索引、缓存与衍生 Wiki 的清理尚未实现（这些组件本身也未实现）。
   - ⚠️ **备份会重新带出已撤下的资料**（设计文档 §15.3）。撤下之后要按组织的保留政策处置既有备份，否则从备份恢复会让已撤下的内容重新暴露。
   - 确认采集遵守目标站点服务条款与访问限制，未绕过技术措施，未高频抓取。
   - 确认正式输出的引用可追溯到具体版本与原文位置，人工审核与错误反馈入口可用。
   - 合规要求不因轻量部署或扩容而降低（设计文档第 19、20 节）。

11. **数据边界与脱敏**（设计文档第 21 节、需求 FR-15）
   - 确认只接入公共法律数据（法律法规、司法解释、公开裁判文书），不把客户资料、案件卷宗或合同导进来。
   - 第一版按单租户、部门级内部工具部署；SaaS 多租户属远期规划，引入前须满足设计文档第 18 节触发条件。
   - 解析产物入库前必须经过脱敏节点，采用**先定位、后脱敏**；实体映射表按最高敏感级别管理，访问权限严于业务数据。
   - 含个人信息的文书原件须登记为受限范围：脱敏只作用于解析产物，原件若对所有登录用户可读，脱敏即形同虚设。
   - 第一版正则识别器只覆盖身份证号、案号、联系方式；**姓名类实体需要 NER，就位前不得把解析产物作为对外发布内容**。
   - 审计与日志只记录元数据与哈希，不记录请求或回答正文。
   - 公共数据表**已**按设计文档 §21.2 去掉组织字段（迁移 0010、0011）；`access_scope='organization'` 现意为"所有已登录用户可读"。

12. **文档解析**（设计文档第 2、6、17.2 节）
   - 第 1 层原生解析已实现（PDF 后端 pypdfium2、pdfplumber 可切换、python-docx、lxml、纯文本）；**第 2 层 OCR 与第 3 层结构化版面（Docling）未实现**。扫描件取不到文本层时标记 `needs_review`，须先补 OCR 才能作为检索来源。
   - PDF 后端**已锁定 pypdfium2**（配置项 `PARSING_PDF_BACKEND` 可切回 pdfplumber）；切换后端会新建解析版本，不移动既有引用。
   - **Docling 暂缓引入**，避免拉取大模型权重造成镜像膨胀与维护成本；待 P4 检索确需表格结构化时再按设计文档第 18 节评估接入。
   - 解析在 16 GB 单机上有硬上限（字节 / 页数 / 字符数）与进程内存增长守卫；正式入库前按设计文档第 14.2 节压测并记录资源占用，优先降并发而不是加交换空间。
   - 印刷页码暂不检测（存 NULL）。
   - **结构识别已实现**（设计文档 §5.2、§6、§7）：识别 编/章/节/条、排除「目录」重复标题，**分块对齐到「条」**并把章节路径写入 `chunks.structure_path`。**款/项/目 未建模**（无编号或依赖款）；**条款身份只覆盖「条」**（章号/节号随上级重置，不能稳定标识）。
   - 解析产物**已关联** `legal_versions` / `provision_versions`（见第 14 项）；**质量检查与人工校正环节仍未实现**——正式作为检索来源前，须先补设计文档 §7 的人工校正与 §20.5 的人工审核。
   - 解析结果在入库前**强制经脱敏节点**（设计文档 §21.3）：`chunks.text` 存**脱敏文本**、原文本不落库（由映射表可逆重建）、`chunk_spans` 偏移仍指向原文本；映射表与分块、审计同事务写入。**不得以"数据已公开"为由跳过脱敏**；映射表按最高敏感级别管理，访问权限须严于业务数据（见第 11 项）。

13. **后台任务 worker**（设计文档 §3.1、§12.2、§14.2）
   - 导入只登记任务；实际解析由**按需启动的单进程 worker** 执行：`python -m app.cli run-worker`（`--once` 只处理一个任务，便于脚本化）。不启动 worker 时任务一直停在 `pending`。
   - worker 是**单进程单并发**、任务结束即释放资源；不要在 16 GB 机器上并发起多个 worker（`worker_concurrency: 1`）。
   - 任务领取原子化（`FOR UPDATE SKIP LOCKED`），有租约与心跳；**worker 被强杀不会丢任务**——租约过期后由回收逻辑退回重试（已用尽重试次数则置 `failed` 进人工处理）。
   - 失败区分确定性（原件缺失 / 超上限 / 解析被拒 → 直接 `failed`）与可重试（原件暂不可用、内存超预算 → 退避重试）；**资源不足记 `resource_exhausted`，不伪装成业务结果**（FR-13）。排查时先看 `jobs.error_code` / `error_message`。
   - 取消只作用于**尚未开始**的任务；正在执行的解析不会被强制终止。
   - 上线前须把 worker 纳入进程管理（开机/异常重启、日志收集）与监控（`pending` 积压、`failed` 数量）。

14. **法律版本落库**（设计文档 §5.1、§5.2、§7、§8.3）
   - 解析成功且识别到法律结构时，会自动把结果挂到法律本体与版本上（`legal_instruments` / `legal_versions` / `provision_identities` / `provision_versions`）；识别不到结构（非法律文本）则跳过。无需额外命令。
   - **版本标识与效力状态来自正文前言的启发式提取**：版本标识取公布信息里的「通过/修正/修订」子句（如 `2018年修正`），取不到才回退文件名 `_YYYYMMDD`；效力状态由「自…起施行」日期推导为「当前有效」/「已公布未生效」，取不到记「未知」。**未知不虚构**。
   - **务必人工复核低置信度数据**：`legal_versions.review_status = 'pending'` 的记录需要确认（版本标识来自文件名回退、标题退化为文件名、或同一版本出现多个原件）。目前**没有待审队列或界面**，只能用 SQL 查 `review_status='pending'` 并结合审计事件 `legal_version.artifact_conflict` 人工处理。
   - **「已公布未生效」的版本不得作为现行依据**（设计文档 §8.3 要求检索默认屏蔽 `legal_status='not_yet_effective'`）；检索层尚未实现，实现时须落实该过滤。
   - 效力状态在落库时计算；时间推进导致的状态变化（如「已公布未生效」到期转有效、旧版本随之被取代）**不会自动更新**，需重新关联或补定期重算。
   - 上线前须复核本体的规范化名称（`title` 是身份键的一部分，改了会新建本体）与 `legal_instruments(jurisdiction, title)` 唯一约束下的既有数据。
   - **择优规则变化后要重挂版本树**：已落库的主原件不会自动重选。`python -m app.cli relink-versions --as kadmin [--instrument <本体 id>...] [--dry-run]` 从解析产物重建、按当前规则重挂（**不重新解析原件、不删数据**，只改 `legal_versions.artifact_id` 与据此重建的条款版本）；需 `source.manage`，操作写审计。**先跑 `--dry-run`**。注意**效力状态优先于格式**：PDF 前言含主席令与「自…起施行」时状态确定，DOCX 若不含主席令则读作 `unknown`，此时 PDF 仍会当选——这是规则本身，不是漏改。

15. **精确检索**（设计文档 §8.1、§8.3、§13）
   - 端点 `POST /api/v1/search`（需 `document.read`）：按名称、文号、稳定 ID、规范化条号精确匹配，叠加法域、资料类型、`effective_on`、`review_status` 过滤。
   - **「已公布未生效」默认不返回**（`include_not_yet_effective` 显式放开）：把尚未生效的法律当作现行依据是最危险的错误。
   - **审核状态不过滤但始终回传**（结果里带 `legal_status` 与 `review_status`）：`pending` 是数据质量标记、不是效力事实，回答层据此判断能否下结论。
   - 授权在数据库中复核（join 原件 + 授权范围），不信任调用方提交的范围；没有原件的版本不参与检索。
   - 条号与文号接受常见写法（`第八十七条` / `87` / `第八十七条之一`；`主席令第七十七号` / `第77号` / 全角数字）；**解析不出来时返回空结果，不会退化成不过滤**。
   - **关键词、向量与图谱检索未实现**（设计文档 §8.2、§8.4 属 P4）；本端点只覆盖 §8.3 的精确字段与过滤。
   - **中文关键词检索已实现**（V1.12）：请求体加 `keyword` 字段，与精确字段和过滤条件 **AND 叠加**。方案经金标准选型（去空白 + 多词 AND，命中后按条文长度排序）。**语料或规模变化时重跑 `backend/scripts/evaluate_retrieval.py`**（只读，报告落 `evaluations/reports/`）；改选型只需改 `app/modules/retrieval/keyword.py` 的 `SELECTED`。⚠️ **PDF 文本里的折行空格**（实测「民用航 空器」）不处理会**丢掉一半召回**——这是采集/解析质量问题，不只是检索细节。当前未建索引（21 ms/查询），条文量上去后再评估 `pg_trgm` 表达式索引，仍不够才按设计文档 §18 考虑 Elasticsearch。**向量与图谱通路仍未实现**。
   - **向量通路与查询改写已接线**（V1.13）：`POST /api/v1/search` 的 **`semantic`** 字段走**级联**——先用关键词（含改写）在原文里找，**命中为空才**用本地嵌入模型走向量兜底；响应里的 **`path`** 告诉调用方这次走的哪条通路（`exact` / `keyword` / `vector`）。**`semantic` 与 `keyword` 互斥**（同时给出报 422）。**不用 RRF 融合**——实测它把关键词准确率从 1.000 拉到 0.035。基础设施：`provision_embeddings` 按模型存向量、`app/adapters/embedding.py` 下模型（走 `HF_ENDPOINT` 镜像的**直链**——**不要用 `hf_hub_download`**，镜像的 `/api/` 返回 403 会落下 0 字节文件）、`scripts/embed_provisions.py` 编码、`app/modules/retrieval/rewrite.py` 改写查询。⚠️ **首次 `semantic` 查询会加载模型**（BGE-M3 约 2 GB 常驻内存、加载约 30 s）。三份金标准在 `evaluations/datasets/`（词面 28 / 问句式 24 / 简称 12），语料或规模变化时重跑 `evaluate_retrieval.py`。⚠️ 换 pgvector 镜像会改 glibc 版本，**必须 `REFRESH COLLATION VERSION` + `REINDEX DATABASE`**。
   - **图谱引用边**（V1.14）：`python -m app.cli extract-citations --as <用户>` 从条文正文抽引用并落成 `provision_relations` 的 `cite` 边（**可重复执行**，冲突行跳过；**逐部法律写审计**）。支持 `--dry-run` 预演。⚠️ 只抽带「本法/本条例」限定词、或法律简称能**唯一**匹配已入库法律名的引用——**裸引用不抽**，所以抽到的条数是下界。新增语料后要重跑。图扩展（`retrieval/graph.py`）**对命中质量零增量**，只作上下文用，**不要接进结果排序**。
   - **Wiki 审核发布**（V1.15）：`PUT /wiki/pages/{id}/revisions/{n}/citations` 登记引用 → `POST .../submit` 提交（**此后该修订锁定**）→ `POST .../publish`（需 `review.decide`）或 `.../reject`。`GET /wiki/revisions/pending` 是待审队列，`GET /wiki/pages/{id}/published` 是读者看到的版本。⚠️ **发布前必须登记引用**（至少一条），且**审核人要对每条引用有访问权**；**作者不能审自己的修订**。⚠️ 引用了受限原件的页面，**看不到原件的人整页都看不到**——这是设计 §10.3 的权限继承，不是 bug。⚠️ 迁移 0016 移除了 `ck_wiki_draft_only`，**回滚会先把非草稿修订改回 draft**。
   - **Wiki 更新提醒**（V1.16）：`python -m app.cli flag-stale-pages --as <用户> [--dry-run]` 扫描**已发布页面**，命中就标记待复核（引用的法律版本被取代、或引用条款正文与发布时快照不一致）。**只标记、不动正文**（§10.2「不自动删除历史说明」）；**重新发布自动清标**；可重复执行，**只在状态变化时写审计**。⚠️ 扫描是**全库**的，不是按组织——多组织部署时按需加 `--organization`（尚未提供）。新增语料、重挂版本树、或重新解析之后都该跑一次。
   - **本地生成模型（P6）**：`python -m app.cli ask --as <用户> --question "…" [--limit N] [--model M]`。⚠️ **运行时是 Ollama**（本项目**只用本地模型**，设计 §9.5 已锁定、默认关闭外部 API）：先 `ollama create legalmind-qwen2.5-7b -f ops/Modelfile.qwen2.5-7b` 把本地 GGUF 导入（**只导入、不重下**），再确认 `OLLAMA_HOST`（默认 `http://127.0.0.1:11434`）。⚠️ **不要在 Python 里直接读 GGUF**——`transformers` 会把整个模型反量化成 fp32，7B 要约 28 GB 内存，普通机器放不下（实测 OOM）；`llama-cpp-python` 在 Windows 上装不了（源码包撞 260 字符路径上限、预编译 wheel 在 GitHub 不可达、自行编译被安全策略封 `reg.exe`），见《技术决策与踩坑记录》§5.1。⚠️ **模型不可用时如实降级**成「只给检索证据」，不回落任何外部服务。生成用温度 0（可复现）。
   - 真实语料冒烟：`监狱法 第一条` 默认 0 命中（其 2026 年修订版为「已公布未生效」），放开后命中；`商标法` 同理。**合成语料测不出条款文本与引用锚点的取法（分块段首刻意含章节标题），必须用真实语料冒烟。**

16. **入库质量门禁**（设计文档 §7、§17.1、§20.3）
   - 只读校验：`set -a && . ./.env && set +a` 并导出 `DATABASE_URL` 后运行 `.venv/Scripts/python.exe scripts/quality_gate.py`（`--source` 限定来源、`--json` 输出 JSON、`--fail-on-degraded` 把降级也算不通过）。退出码 1 表示门禁不通过；报告落在 `evaluations/reports/`。
   - 三态：`failed`（来源未登记 / 授权说明为空 / 无解析产物 / 未挂法律版本）、`degraded`（`review_status='pending'`、版本标识未标注、无条款身份）、`passed`（高置信度自动落库且条款齐全）。
   - **门禁只校验、不阻断**：`failed` 不会自动把数据撤出（撤下走第 10 项的 `withdraw-document`）。
   - 真实语料当前 15 passed / 4 degraded / 1 failed；`failed` 的是纯图像扫描件的宪法 PDF（解析 0 字符），需补第 2 层 OCR 才能入库。
   - **`approved` 是自动确认，不等于人工复核**；`pending` 需人工确认，目前没有待审队列或界面（见第 14 项）。

**这就是完整统一命名后的 LegalMind 开发脚手架：有实际前后端、迁移和 Wiki 修订闭环，同时明确保留法律知识系统最重要的权限、版本和证据核验扩展边界。**
