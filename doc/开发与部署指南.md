from uuid import UUID

from fastapi import HTTPException
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.security import Principal
from app.models import (
    AuditEvent,
    OutboxEvent,
    WikiPage,
    WikiRevision,
)
from app.modules.wiki.schemas import CreatePage, CreateRevision


def record_change(
    session: AsyncSession,
    principal: Principal,
    page_id: UUID,
    revision: WikiRevision,
    action: str,
) -> None:
    payload = {
        "page_id": str(page_id),
        "revision_id": str(revision.id),
        "revision_number": revision.number,
    }

    session.add(
        AuditEvent(
            organization_id=principal.organization_id,
            actor_id=principal.user_id,
            action=action,
            resource_id=page_id,
            payload=payload,
        )
    )

    session.add(
        OutboxEvent(
            organization_id=principal.organization_id,
            event_type=action,
            payload=payload,
        )
    )


async def create_page(
    session: AsyncSession,
    principal: Principal,
    data: CreatePage,
) -> WikiRevision:
    async with session.begin():
        page = WikiPage(
            organization_id=principal.organization_id,
            title=data.title,
            head_revision=1,
        )
        session.add(page)
        await session.flush()

        revision = WikiRevision(
            page_id=page.id,
            number=1,
            body=data.body,
            author_id=principal.user_id,
        )
        session.add(revision)
        await session.flush()

        record_change(
            session,
            principal,
            page.id,
            revision,
            "wiki.page.created",
        )

        await session.flush()
        await session.refresh(revision)

    return revision


async def create_revision(
    session: AsyncSession,
    principal: Principal,
    page_id: UUID,
    data: CreateRevision,
) -> WikiRevision:
    async with session.begin():
        page = await session.scalar(
            select(WikiPage)
            .where(
                WikiPage.id == page_id,
                WikiPage.organization_id == principal.organization_id,
            )
            .with_for_update()
        )

        if page is None:
            raise HTTPException(
                status_code=404,
                detail="Page not found",
            )

        if page.head_revision != data.expected_revision:
            raise HTTPException(
                status_code=409,
                detail="Revision conflict: reload before editing",
            )

        page.head_revision += 1

        revision = WikiRevision(
            page_id=page.id,
            number=page.head_revision,
            body=data.body,
            author_id=principal.user_id,
        )
        session.add(revision)
        await session.flush()

        record_change(
            session,
            principal,
            page.id,
            revision,
            "wiki.revision.created",
        )

        await session.flush()
        await session.refresh(revision)

    return revision
''',

PrincipalDep = Annotated[Principal, Depends(get_principal)]


@router.get("/pages", response_model=list[PageOut])
async def list_pages(
    session: SessionDep,
    principal: PrincipalDep,
    before: UUID | None = None,
    limit: Annotated[int, Query(ge=1, le=100)] = 50,

    statement = select(WikiPage).where(
        WikiPage.organization_id == principal.organization_id

    return list(result)

    headers.set("X-Dev-Key", key);

    if (init.body !== undefined) {
      headers.set("Content-Type", "application/json");
    }

    const response = await fetch(`/api/v1${path}`, {
      ...init,
      headers,
    });

      throw new Error(
        `${response.status}: ${
          payload?.detail
            ? JSON.stringify(payload.detail)
            : response.statusText

# 七、正式部署前必须修改

   - 部署在反向代理后时配置 TRUSTED_PROXIES，并确认代理会覆盖而不是透传客户端的 X-Forwarded-For。
   - 页面级授权目前只覆盖 Wiki 页面；后续的来源、法律版本、检索和问答须接入同样的对象级授权。

2. **区分数据库账号**
   - 迁移账号与业务账号分离。
   - 当前 Compose 使用的是开发账号，不是生产最小权限配置。

3. **锁定依赖**（**已完成**，V1.39；跑法见 `CONTRIBUTING.md`「依赖锁定」）

   - ✅ **镜像**：`compose.yaml` 的 postgres / redis 固定了 `sha256` 摘要（取法与「按平台取、换 arm64 要重取」见 CONTRIBUTING）。

4. **替换前端开发服务器**
   - 正式环境使用构建产物及反向代理。
   - 配置 HTTPS、请求大小限制和超时。

5. **完善审计**
   - 当前只是事务内记录。
   - 尚无数据库级 append-only 权限、归档和外部校验点。

6. **完善任务可靠性**
   - Outbox 表不等于可靠任务系统。
   - 需要投递器、幂等消费、重试、超时和对账。

7. **补集成测试和恢复演练**
   - 尤其是跨组织访问、事务回滚、重复任务、数据库恢复。

8. **评估图谱存储**
   - GraphRAG 概念验证阶段基于 PostgreSQL + pgvector，不新增常驻组件。
   - 迁移 Neo4j 前，须完成准确率评测并满足设计文档第 18 节触发条件，同时补充图存储的备份、恢复和权限隔离方案。

9. **真实资料入库前先做权限与来源**
   - 不要先把客户资料导进去，再补权限与外发控制。

10. **合规前置检查**（设计文档第 20 节、需求 FR-14）
   - 确认系统定位为法律信息辅助工具；页面、导出物与问答输出均带“不构成法律意见”声明与知识范围说明。
   - 确认每个来源均已登记授权说明（`license_note`）与最后核查时间；官方渠道优先，非官方来源有授权依据。
   - **逐份核对原件的来源归属**，不能只看文件标题——同一批文件可能来自不同来源（官方库与第三方数据库并存）。归属登记有误时用 `PUT /api/v1/documents/{id}/source` 或 `app.cli set-document-source`（需 `source.manage`）更正；来源登记信息本身有误时用 `PUT /api/v1/sources/{id}` 或 `app.cli update-source`。两者都写审计（含变更前后值），因为设计文档 §20.3 要求「来源更新、撤回或内容变化须留痕」。
   - ⚠️ **版权声明不是授权说明**。第三方数据库页脚常见的「版权所有：某公司 + ICP 备案号」只表明该库主张权利，**不构成我们存储与使用的授权**。授权说明必须写明本机构的使用依据（官方公开、订阅协议、书面许可等）与使用范围；**未取得授权的来源，其内容不得入库**（§20.3）。**不得为了通过检查而填写与事实不符的授权说明**——审计会记录每次变更前后的原文。
   - **来源被认定无权使用时须撤下其原件**：`python -m app.cli withdraw-document --as kadmin --from-source <来源 ID> --reason "原因"`（先加 `--dry-run` 预演）。撤下会删除原件、解析版本、分块、定位与脱敏映射（设计文档 §15.3 要求删除覆盖原件与解析产物），**不可逆**。同一版本若还有别的合法来源原件，版本树会从它重建，不会被连带删除。索引、缓存与衍生 Wiki 的清理尚未实现（这些组件本身也未实现）。
   - ⚠️ **备份会重新带出已撤下的资料**（设计文档 §15.3）。撤下之后要按组织的保留政策处置既有备份，否则从备份恢复会让已撤下的内容重新暴露。
   - 确认采集遵守目标站点服务条款与访问限制，未绕过技术措施，未高频抓取。
   - 确认正式输出的引用可追溯到具体版本与原文位置，人工审核与错误反馈入口可用。
   - 合规要求不因轻量部署或扩容而降低（设计文档第 19、20 节）。

11. **数据边界与脱敏**（设计文档第 21 节、需求 FR-15）
   - 确认只接入公共法律数据（法律法规、司法解释、公开裁判文书），不把客户资料、案件卷宗或合同导进来。
   - 第一版按单租户、部门级内部工具部署；SaaS 多租户属远期规划，引入前须满足设计文档第 18 节触发条件。
   - 解析产物入库前必须经过脱敏节点，采用**先定位、后脱敏**；实体映射表按最高敏感级别管理，访问权限严于业务数据。
   - 含个人信息的文书原件须登记为受限范围：脱敏只作用于解析产物，原件若对所有登录用户可读，脱敏即形同虚设。
   - 第一版正则识别器只覆盖身份证号、案号、联系方式；**姓名类实体需要 NER，就位前不得把解析产物作为对外发布内容**。
   - 审计与日志只记录元数据与哈希，不记录请求或回答正文。
   - 公共数据表**已**按设计文档 §21.2 去掉组织字段（迁移 0010、0011）；`access_scope='organization'` 现意为"所有已登录用户可读"。

12. **文档解析**（设计文档第 2、6、17.2 节）
   - 第 1 层原生解析已实现（PDF 后端 pypdfium2、pdfplumber 可切换、python-docx、lxml、纯文本）；**第 2 层 OCR 与第 3 层结构化版面（Docling）未实现**。扫描件取不到文本层时标记 `needs_review`，须先补 OCR 才能作为检索来源。
   - PDF 后端**已锁定 pypdfium2**（配置项 `PARSING_PDF_BACKEND` 可切回 pdfplumber）；切换后端会新建解析版本，不移动既有引用。
   - **Docling 暂缓引入**，避免拉取大模型权重造成镜像膨胀与维护成本；待 P4 检索确需表格结构化时再按设计文档第 18 节评估接入。
   - 解析在 16 GB 单机上有硬上限（字节 / 页数 / 字符数）与进程内存增长守卫；正式入库前按设计文档第 14.2 节压测并记录资源占用，优先降并发而不是加交换空间。
   - 印刷页码暂不检测（存 NULL）。
   - **结构识别已实现**（设计文档 §5.2、§6、§7）：识别 编/章/节/条、排除「目录」重复标题，**分块对齐到「条」**并把章节路径写入 `chunks.structure_path`。**款/项/目 未建模**（无编号或依赖款）；**条款身份只覆盖「条」**（章号/节号随上级重置，不能稳定标识）。
   - 解析产物**已关联** `legal_versions` / `provision_versions`（见第 14 项）；**质量检查与人工校正环节仍未实现**——正式作为检索来源前，须先补设计文档 §7 的人工校正与 §20.5 的人工审核。
   - 解析结果在入库前**强制经脱敏节点**（设计文档 §21.3）：`chunks.text` 存**脱敏文本**、原文本不落库（由映射表可逆重建）、`chunk_spans` 偏移仍指向原文本；映射表与分块、审计同事务写入。**不得以"数据已公开"为由跳过脱敏**；映射表按最高敏感级别管理，访问权限须严于业务数据（见第 11 项）。

13. **后台任务 worker**（设计文档 §3.1、§12.2、§14.2）
   - 导入只登记任务；实际解析由**按需启动的单进程 worker** 执行：`python -m app.cli run-worker`（`--once` 只处理一个任务，便于脚本化）。不启动 worker 时任务一直停在 `pending`。
   - worker 是**单进程单并发**、任务结束即释放资源；不要在 16 GB 机器上并发起多个 worker（`worker_concurrency: 1`）。
   - 任务领取原子化（`FOR UPDATE SKIP LOCKED`），有租约与心跳；**worker 被强杀不会丢任务**——租约过期后由回收逻辑退回重试（已用尽重试次数则置 `failed` 进人工处理）。
   - 失败区分确定性（原件缺失 / 超上限 / 解析被拒 → 直接 `failed`）与可重试（原件暂不可用、内存超预算 → 退避重试）；**资源不足记 `resource_exhausted`，不伪装成业务结果**（FR-13）。排查时先看 `jobs.error_code` / `error_message`。
   - 取消只作用于**尚未开始**的任务；正在执行的解析不会被强制终止。
   - 上线前须把 worker 纳入进程管理（开机/异常重启、日志收集）与监控（`pending` 积压、`failed` 数量）。

14. **法律版本落库**（设计文档 §5.1、§5.2、§7、§8.3）
   - 解析成功且识别到法律结构时，会自动把结果挂到法律本体与版本上（`legal_instruments` / `legal_versions` / `provision_identities` / `provision_versions`）；识别不到结构（非法律文本）则跳过。无需额外命令。
   - **版本标识与效力状态来自正文前言的启发式提取**：版本标识取公布信息里的「通过/修正/修订」子句（如 `2018年修正`），取不到才回退文件名 `_YYYYMMDD`；效力状态由「自…起施行」日期推导为「当前有效」/「已公布未生效」，取不到记「未知」。**未知不虚构**。
   - **务必人工复核低置信度数据**：`legal_versions.review_status = 'pending'` 的记录需要确认（版本标识来自文件名回退、标题退化为文件名、或同一版本出现多个原件）。
   - ✅ **复核入口（V1.42 CLI / V1.43 界面）**：
     - 界面：`#/versions`（「版本复核」，入口按 `review.decide`）——列待审、逐条「确认可用 / 驳回」，备注写进审计。
     - CLI：`python -m app.cli pending-versions --as <用户>` 列待审（带法律名称与主原件），`python -m app.cli review-version --as <用户> --version <版本 id> --approve|--reject [--note]` 出结论。
     - 两者都需 `review.decide`，都写审计事件 `legal_version.reviewed`。⚠️ **只改 `review_status`，不动 `legal_status`**——后者是法律事实，不是审核意见。⚠️ **可以改已复核的版本**（审错了要能纠正），重复同一结论报 409。
   - 结合审计事件 `legal_version.artifact_conflict`（落库时的多原件冲突）一起判断。
   - **「已公布未生效」的版本不得作为现行依据**（设计文档 §8.3 要求检索默认屏蔽 `legal_status='not_yet_effective'`）；检索层尚未实现，实现时须落实该过滤。
   - ✅ **时间推进导致的漂移已可重算（V1.45）**：`python -m app.cli recompute-statuses --as <用户> [--today YYYY-MM-DD] [--dry-run]`（需 `source.manage`，写审计与 Outbox）。把「已公布未生效」且施行日期已到的版本转为现行有效，并让公布更晚的版本取代更早的。
   - ⚠️ **这是维护动作，不是定时任务**：本项目不引入调度器。**每次导入新语料后跑一次**，以及在有版本临近施行日期时跑一次。`--dry-run` 预演、`--today` 可以给未来日期（`--today 2026-11-15` 就能看到「那天会发生什么」）。

   - 上线前须复核本体的规范化名称（`title` 是身份键的一部分，改了会新建本体）与 `legal_instruments(jurisdiction, title)` 唯一约束下的既有数据。
   - **择优规则变化后要重挂版本树**：已落库的主原件不会自动重选。`python -m app.cli relink-versions --as kadmin [--instrument <本体 id>...] [--dry-run]` 从解析产物重建、按当前规则重挂（**不重新解析原件、不删数据**，只改 `legal_versions.artifact_id` 与据此重建的条款版本）；需 `source.manage`，操作写审计。**先跑 `--dry-run`**。注意**效力状态优先于格式**：PDF 前言含主席令与「自…起施行」时状态确定，DOCX 若不含主席令则读作 `unknown`，此时 PDF 仍会当选——这是规则本身，不是漏改。⚠️ **2026-10-08 用户已拍板维持现状**（不做「docx 一律优先」）：拿解析便利性换法律效力确定性不划算，要处理那 5 个版本应当**补数据**（人工修正 `legal_status`，或从 PDF 前言抽主席令回填 DOCX 元数据后再 `relink-versions`），**不是改规则**。裁决记录见设计文档第 11 节。

15. **精确检索**（设计文档 §8.1、§8.3、§13）

17. **生成质量评测**（设计文档 §9.5「本地生成模型单独评测，达标后再接入正式问答」）
   - 金标准：`evaluations/datasets/generation_quality.json`（22 条）。**每条用例显式钉住要喂给模型的条款**，评测时**绕过检索**——量到的只有生成质量，检索另有三份金标准。
   - 跑法（需要本地模型；Ollama 在跑且模型已导入）：
     - 只校验数据集（几秒，不加载模型）：`--check-only`。它会核对每条 `required_facts` / `forbidden_facts` 确实是条款正文的**去空白子串**、`expect_citations` 是 `evidence` 的子集——**期望是推导的、可复核的**，不是手写的参考答案。
     - 全量：`.venv/Scripts/python.exe scripts/evaluate_answering.py`（22 条 × 约 5–15 s）。冒烟用 `--ids g01,a01`；当门禁用 `--gate`（未达标退出码 1）。报告落 `evaluations/reports/`。

   - **效力状态提示不在模型指标里**（V1.19）：那是回答层的职责（§8.3），报告里作**诊断项**照记（当前 0.000，即模型不会主动说）。回答层的强制提示在 `answering/service.py`：依据里有非现行有效版本就给 `status_notice`，**全部非现行有效则直接拒答、不生成结论**；⚠️ **`unknown` 不算**非现行有效（只表示没提取到施行日期），否则这份语料 13 部法律有 7 部会被判成不可用。

18. **效力状态门禁**（设计文档 §8.3、§9.5）
   - **由回答层确定性强制，不问模型**：`python -m app.cli ask` 时，若依据里有非现行有效版本（`repealed` / `not_yet_effective`），会在答案前打印 `⚠️ 依据提示：…`；**若全部依据都不是现行有效，直接拒答并只列出条文，不生成结论**（§9.5「不开放正式自动结论」），此时不会调用本地模型。
   - **为什么不由提示词解决**：实测（第 17 项）证据块里明写「（尚未生效，不得作为现行依据）」，模型照样把未生效的法律当现行依据陈述。安全属性不能交给模型自觉——这与本系统「授权、证据绑定、审计由本系统实现，不依赖框架与模型隐式行为」是同一条口径。
   - ⚠️ **`unknown` 不算**非现行有效：它只表示没提取到施行日期（提示词里也是这么告诉模型的），算进来会把这份语料 13 部法律里的 7 部判成不可用。
   - ⚠️ **没有「就是要查已废止旧法」的显式开关**，调用方无法主动放开这条拒答。检索默认只屏蔽 `not_yet_effective`（§8.3），`repealed` 版本会正常返回，所以这条门禁在真实链路上主要防的是**引用到已被取代的旧版本**。

19. **引用核验（§9.3 第一层）**（设计文档 §9.3、§9.4）
   - 位置：`app/modules/answering/verification.py`（纯函数，无 IO）。在 `answer_question` 里于生成之后、返回之前跑；**没过就不当正式答案发布**——`answer` 变成「未通过核验」说明，模型草稿在 `Answer.draft` 里，`Answer.published` 为假。
   - **核验的是模型写出来的依据**（返回给调用方的 `citations` 来自检索、天然在授权范围内，不会凭空出现）：① 指名引用（法名 + 同一句内的条号）必须落在证据集内；② 条号必须是证据条款本身或**证据正文里出现过的条号**（复述条文里的交叉引用不算编造）；③ 阿拉伯数值必须来自证据或问题；④ 引号里的引文必须能在证据正文里找到。
   - ⚠️ **只提法名不引条号不算引用**——那通常是在说明「缺什么」（实测 22 条里唯一一条误报就是「依据不足……（《道路交通安全法》未提供相关条款）」）。
   - ⚠️ **法名用简称全等比对**，不用子串：「药品管理法」会被「药品管理法实施条例」掩盖（`legal_corpus/citations.py` 踩过这个前缀坑）。
   - **触发率**：评测脚本会把它作为诊断项 `verification_failure_rate` 报出来（当前 22 条上为 **0**）。**门禁本来就不该常响**——它涨上去说明模型在编依据，或者规则定得过严。
   - ⚠️ **只做第一层**：第二层语义校验（证据是否支持主张、有无过度概括）未实现；「证据是否发布/可访问/未被撤回」由检索的授权复核与 `withdraw-document` 保证，不在这里重复。数值检查只看**阿拉伯数字**，中文数字写法（「二十年」）不查。

20. **证据上下文预算装配**（设计文档 §8.2 第 9 步、§9.4）
   - 位置：`app/modules/answering/assembly.py`（纯函数）。在生成之前跑，**决定把哪几条证据放进提示**。
   - **为什么必须自己算**：检索只限条数（`limit`）不限长度；装不下模型窗口时推理引擎**不报错，而是静默从前面截断**——指令就在提示最前面（指令与证据在同一条用户消息里），于是**模型先失去全部约束**。这正是设计 §9.4 禁止的「无提示截断」。
   - 两条规则：**整条进或整条不进**（截一半的条文读起来仍像完整的，比不装更危险）；**装不下就如实说**（`TOO_LARGE_NOTICE` 不生成结论；`PARTIAL_NOTICE` 出结论但限定范围并点名未装入的条文，落在 `Answer.evidence_notice`，CLI 打印）。
   - ⚠️ **窗口要改两处**：`GENERATION_CONTEXT_TOKENS`（默认 8192）与 `ops/Modelfile.qwen2.5-7b` 的 `PARAMETER num_ctx` 必须一致，否则预算与实际窗口对不上。
   - ⚠️ **预算按字符估**（`CHAR_PER_TOKEN = 1`），因为本项目不持有模型 tokenizer。**估多了只是少装一条，估少了会退回静默截断**——换模型后要按实际重新核这个系数。
   - ⚠️ **`_cost` 直接量 `build_evidence` 的输出**，不要另写一份排版：首版自己写了一份、漏了效力状态那段，**估少了两格**（被单元测试当场抓到）。
   - **当前余量**：22 条金标准的证据块最大 598 字、占预算 8%，所以这道门禁基本不触发——它是为 `limit` 调大或检索到法典长条文准备的。

21. **输出边界与合规声明**（设计文档 §20.2、§9.3 第三层）
   - **每个正式输出都带声明**：`Answer.disclaimer`（「不构成法律意见 + 依据限于本次检索到的条文」）与 `Answer.generated_at`（ISO 8601）是**默认字段**——任何返回路径（含拒答）都带着它们，**结构上发布不出一份不带声明的结论**。CLI 在末尾打印。
   - **不对个人情形下结论**：`app/modules/answering/scope.py` 的 `detect_personal_scope()` 识别「第一人称 + 资格/权利义务判断」的提问（「我能否」「我该不该」「我是否符合」），命中即 `Answer.blocked_by="scope"`、**不生成个人结论**，只回条文与下一步。
   - **为什么**：实测问「我是名特困人员，我能否领到社会救助？」，模型引用正确、内容也对，却补了一句「因此，作为特困人员，可以领取社会救助」——「法律怎么规定」变成「你能领」，正是「无资质提供法律服务」那条风险（设计 §20.1 风险映射表），而且它跳过了「特困人员须经认定程序」这个前提。
   - ⚠️ **识别是规则匹配，不是语义理解**：会漏（「像我这情况能不能领」就绕过去了）也会误报。后果刻意定得很轻——**不给个人结论、把条文交回人工**，不是「判定用户不能问」。**宁可多转一次人工，也不要多给一句个人结论。**
   - **`Answer.published`** 覆盖**两道**门禁：提问范围（`scope`）与引用核验（`verification`）。为假即不得作为正式答案。
   - ⚠️ **没有真正的待审队列与界面**：`published` 只是个标志，CLI 仍会打印未发布的结论（已标注「不作为正式答案」）。「输出须标注所用证据版本与不确定性」只做了一半——版本逐条给了，**不确定性没有量化表达**。声明目前只在 CLI 打印，前端与导出物未接入。

22. **结构化输出（§9.2）**（设计文档 §9.2、§9.3 第一层）
   - 位置：`app/modules/answering/claims.py`。模型只输出 `claims`（含 `text` / `evidence_ids` / `conditions` / `limitations`）与 `missing_facts` / `conflicts`；**`evidence_ids` 就是证据块里的编号 `[1]` `[2]`**。
   - **引用由服务端渲染**：模型不再写「依据：《某法》第某条」，`claims.render()` 从编号查标题拼进正文。**证据块的排版与编号因此是契约的一部分，不能随便改。**
   - **核验**：`verification.verify_claims` 判 `evidence_ids` 在不在证据编号内、**每条主张是否都挂了编号**（没有编号的主张就是没有依据）。文本层检查（法名 / 条号 / 数值 / 引文）仍然跑，因为主张正文本身也可能写出证据外的条号。
   - ⚠️ **生成必须传完整 JSON Schema**：`generation.generate(..., schema=claims.SCHEMA)`。**只用 `format: "json"` 不行**——实测 22 条里 **11 条**输出 `"limitations[]`（键与空数组之间漏冒号，且只在空数组时发生），换 schema 后同一批 7/7 通过、全量解析失败率 **0.500 → 0.000**。**「请输出 JSON」是请求，「只能输出这个 JSON」是语法。**
   - ⚠️ **容错解析会掩盖模型的规矩程度**：`claims.parse` 能救回「多段 JSON 并列」这类残缺，所以 `ParsedAnswer.repaired` 与评测的 `format_failure_rate` 要**一起看**——只报「最终解析成功」会让人以为模型很规矩。
   - ⚠️ **没按约定输出就不当正式答案发布**：`Answer.blocked_by = "format"`，原始输出留在 `Answer.draft` 交人工。
   - ⚠️ **改输出格式必须重跑评测**：结构化主张更简洁，模型一度丢掉主体与适用范围（「国务院应当做什么」答成「组织对国家发展规划实施情况进行中期评估」），`fact_coverage` 掉到 0.765；提示里补一句「每条主张要写完整，带上主体与适用范围」后回到 0.912。
   - **未做**：第二层语义校验（`evidence_ids` 只在编号内校验，**不判断该条文是否真的支持该主张**）；`conflicts` 只渲染不处理。
23. **第二层语义校验（已建，未接入）**（设计文档 §9.3 第二层）
   - 位置：`app/modules/answering/semantics.py`（判官）+ `evaluations/datasets/claim_verification.json`（断言级金标准，28 条）+ `backend/scripts/evaluate_semantics.py`（度量）。跑法：`--ids` 只跑几条、`--gate` 未达标退非零；报告落 `evaluations/reports/`。
   - **它做什么**：逐条主张问本地模型「这些条文是否支持这条主张」，输出 `supported` + `evidence_quote`（依据原句）+ `issues`。
   - **两道确定性核对**（加在语义判断之上）：① 判官交的原句必须真的在条文里（容忍抄上《法名》第X条），**说支持却引不出原文就降级为不支持**；② **「可以」与「应当」不能被互换**（把裁量说成义务是实质性错误）。
   - ⚠️ **未接入问答链路**：金标准实测**缺陷检出率 0.857 < 0.9**，所以它**不拦结论**，只作为适配层与评测对象存在（与重排序同样处置）。要接入得先把这一层做到达标，或者换更强的判官模型。
   - ⚠️ **已知漏判**：把范围放大（「所有」「均」）与抹掉例外（「任何情况下都」「一律」）。
   - ⚠️ **留出集已经用掉一次**（前 21 条调到 21/21、留出 7 条 5/7）——**再调提示词就得再造一批新的留出用例**，否则尺子就废了。

24. **执行状态机与运行记录**（设计文档 §9.1、§9.3 第三层、§9.4）
   - **每个出口都声明状态**：`_answer_pipeline` 的每条返回路径都先 `run.to(<状态>)`；不在转移表（`app.models.ANSWER_RUN_TRANSITIONS`）里的转移会抛 `IllegalTransition`。**加新分支时别忘了声明状态**，否则会当场报错——这是设计意图，不是麻烦。
   - **运行记录**（`answer_runs`）：状态、证据引用、配置快照（含**提示词哈希**）、门禁、是否发布、耗时、复核字段。**没有问题和回答正文**（只留 sha256）——复核时想回看正文，目前只能靠日志或草稿。
   - **待审队列**：`python -m app.cli answer-runs --as <用户> --pending` 列出；`review-run --as <用户> --run <id> --note '…'` 复核（需 `review.decide`，写审计事件 `answer_run.reviewed`）。
   - ⚠️ **复核不改变运行状态**——状态机记的是「当时怎么走的」，复核是之后发生的另一件事，结论写在 `review_note`。重复复核会被拒绝。
   - ⚠️ **转移表里两条刻意的声明**：`RETRIEVING → ASSEMBLING_EVIDENCE`（跳过 `RERANKING`，重排序没进默认）与 `GENERATING → NEEDS_REVIEW`（输出不合约定是「有草稿但需人工」，不是「失败」）。改这两条前先读设计 §9.1 的说明。
   - ⚠️ **`CLARIFYING` / `RERANKING` 是保留状态**（反问与重排序未接入）；**异步 answer-run 见第 25 项**。
25. **异步 answer-run 与短期内容缓存**（设计文档 §9.4、§9.3 第三层、§14.1、§21）
   - **跑起来**：`python -m app.cli ask --as <用户> --question '…' --async` 提交 → `python -m app.cli run-worker --once` 执行 → `python -m app.cli answer-run --as <用户> --run <id>` 查看。三步都要能连上 `CACHE_URL`（宿主机开发时指向 `redis://127.0.0.1:6379/0`）。
   - **内容在哪**：运行记录里**没有正文**（只留 sha256）。问题与回答都在**短期缓存**里，`answer-run` 从那里取；**TTL 到期即不可复核**（默认 24 小时），这是「不落库」的必然代价。
   - ⚠️ **Redis 容器必须关掉持久化**（`--save "" --appendonly no`）：缓存里是「跑过脱敏的回答」，开了 RDB/AOF 就会落盘、进而进备份，**等于变相落库**。另设 `--maxmemory 64mb --maxmemory-policy allkeys-lru` 兜住容量。
   - ⚠️ **留空 `CACHE_URL` 即禁用**：异步问答会**拒绝提交**（提交方靠缓存取结果），同步问答照常；待审队列如实显示「内容不可用」，不会静默留空。
   - ⚠️ **异步的问题先脱敏**（它必须经缓存传给 worker），所以异步与同步的检索输入**不一致**——脱敏只替换身份证/案号/电话，语料里没有这些，不影响检索，但**换语料时要重新评估**。
   - ⚠️ **运行记录与任务是两套状态**：`answer-run` 会同时显示任务状态与尝试次数。**永久失败时运行记录停在最后一次尝试的状态**（不会自动置 `FAILED`——否则重试会撞上终态），所以**卡住要看任务状态**。
   - ⚠️ **没有并发与排队上限**：`--async` 可以无限提交，worker 单进程串行处理。
26. **问答 API 与 SSE 流式进度**（设计文档 §9.1、§9.4、§13）
   - **端点**：`POST /api/v1/answers`（提交异步问答，202；**缓存未配置返回 503**）、`GET /api/v1/answers?pending=true`（待审队列）、`GET /api/v1/answers/{id}`（状态 + 内容）、`GET /api/v1/answers/{id}/events`（SSE）、`POST /api/v1/answers/{id}/review`（复核，重复复核 409）。
   - **SSE 事件**：`state`（每次状态转移）→ `answer`（**只在终态**；门禁未过的推拒答说明、`published=false`）→ `done`；超时推 `timeout` 后结束。超时上限由 `ANSWER_STREAM_TIMEOUT_SECONDS`（默认 300 秒）控制。
   - ⚠️ **错误响应体是 `{code, message, trace_id}`**（全局错误处理器统一过），**不是** FastAPI 的 `detail`——客户端按 `code`/`message` 解析。
   - ⚠️ **测试不能证明「逐条到达」**：httpx 的 `ASGITransport` 会把响应缓冲到生成器跑完（实测首事件延迟 1.5 秒）。要验增量交付得对着真实 uvicorn 跑，并用 `curl -N` 或 `httpx.stream` 打时间戳。
   - ⚠️ **终态先写、结果后写**：终态由链路写、结果与缓存内容随后才落，所以 `answer` 事件前有一个**有界宽限**（`CONTENT_GRACE_POLLS`，默认 5 秒）。这是补丁不是重试——缓存写失败时 5 秒后仍会推 `content_available=false`。
   - ⚠️ **SSE 是「轮询数据库再推送」**，不是事件总线：每个连接各自轮询（`ANSWER_STREAM_POLL_SECONDS`，默认 0.5 秒），**连接数没有上限**。
   - ⚠️ **流式端点不用请求级会话**（一条 SSE 可能挂几分钟），每次轮询开一个短会话。
27. **澄清分支（CLARIFYING）**（设计文档 §9.1、§9.4；需求 FR-06 第 2、3 步）
   - **触发**：问题太笼统时先反问，而不是硬答。判据**确定性**：剥掉疑问框架与空代词后实质内容不足 3 字。⚠️ **「工资怎么办？」被判需要澄清是有意的**（没说工资的什么）；「拖欠工资」直接检索。
   - ⚠️ **判据是规则不是语义防线**：「我想问一下那个事情」这类会漏过去。**漏报的后果只是多问一句，不是拒答**；用户回一句就能继续。
   - **续跑**：`clarify-run --as <用户> --run <运行 id> --answer '<补充>'`（或 `POST /api/v1/answers/{id}/clarify`）→ 再跑一次 `run-worker`。补充会与**原问题合并**（都先脱敏）后继续**同一个运行**。
   - ⚠️ **原问题必须还在短期缓存里**（TTL 默认 24 小时）——过期就只能重新提问。
   - ⚠️ **同一运行只允许一份补充在飞**：连点两次第二次会 409（判据是 `jobs` 表，不是运行状态）。
   - ⚠️ **`CLARIFYING` 的运行不进待审队列**（它在等用户，不是等人判读），**也没有机制提醒「有澄清没人回」**。
   - ⚠️ **同步 `ask` 也依赖缓存**：不传缓存的话澄清没法续跑（CLI 已自动传）。

29. **多轮追问（会话）**（设计文档 §9.6；需求 FR-16）
   - **做什么**：同一会话内的追问继承上文（「那如果是这样呢？」），用户不必重复已说过的条件。
   - **用法**：`POST /api/v1/answers` 带可选 `session_id`；或 CLI `ask --session <uuid> --question '…'`（同步与 `--async` 都支持）。**留空即单轮**。
   - **界面**：问答页页首有**会话栏**（已问几轮 + 「开始新会话」）；运行详情页**终态**出现「继续追问」卡片（会话轮次可点回 + 追问框，提交后新开一次运行）。**会话 id 存 `sessionStorage`，不进 URL、不进 Cookie**；轮次列表**只存在浏览器本地**，换设备看不到（§5.7 已记）。
   - **怎么工作**：`answering/session.py::compose_query` 把**最近 1 轮**的脱敏问题与本次追问拼成**自足的问题**，之后完全走单轮链路（检索 → 生成 → 核验 → 门禁）。`Answer.question` 记**用户实际敲的那句**，`Answer.resolved_question` 记**实际交给检索与模型的那句**——分开记是为了可追溯。
   - ⚠️ **三条边界（不可放开）**：① **历史「答案」绝不进提示词**——答案含结论，进了提示词就会被模型当依据用，而它未经本轮授权复核、未绑条款版本；进提示词的只是**补全后的「问题」**（那是用户输入，不构成依据）。② **会话历史是客户数据 → 只进短期缓存**（`conv:{session_id}` 存脱敏后的问题与证据引用、**不存答案正文**、TTL 与答案缓存同值、过期即焚），**主库不新增会话表**。③ **每条追问是独立运行、各自过门禁**，会话只是分组、**本身不构成正式输出**。
   - ⚠️ **需要 `CACHE_URL`**：缓存未配置时多轮不可用，但**单轮照常**——单轮是主链路，不能因为会话功能而不可用。
   - ⚠️ **不做**：指代消解（规则不可靠、模型推断违反边界①）、会话列表/管理界面、**会话历史落库**（是「不做」不是「暂缓」）。

30. **模型不可用的兜底（探测得到 ≠ 用得了）**（设计文档 §9.4、§9.5）
   - **背景**：§9.5 写着「模型不可用就如实降级成『只给证据 + 转人工』、不回落外部服务」，但兜底原先只在**入口**做——`generation.available()` 查 Ollama 的 `/api/tags`。
   - ⚠️ **`/api/tags` 证明不了 runner 能用**：实测 v0.40.0 自动更新事故里 `llama-server binary not found`、模型调用**统一 500**，而 `ollama list` 与 `/api/tags` **完全正常**。「探测通过 → 真正调用失败」于是以一次 500 收场。
   - **改法**（V1.36）：`generation.generate()` 把传输失败（`URLError` / `HTTPError` / 超时 / 非 JSON 响应）统一包装成 `GenerationUnavailable`；回答层捕获后与「探测为假」走**完全相同**的降级路（抽成同一个出口，两处语义不许分叉），澄清分支捕获后如实退回模板问法。
   - ⚠️ **`GenerationUnavailable` 不是 `OSError` / `URLError` 子类**：澄清分支那条 `except` 必须**显式认它**，否则模型一挂，澄清就从「退回模板」变成一次 500。
   - **怎么验**：`tests/test_generation.py`（5 项，含「正常路径不受影响」）；`tests/test_answering.py` 的 `test_unavailable_model_degrades_to_evidence_only` / `test_generation_failure_degrades_instead_of_raising`；`tests/test_clarify.py` 的 `test_generation_unavailable_also_falls_back_to_the_template`。

31. **模型缓存目录的就绪判据（半截目录要能自愈）**（设计文档 §8.2）
   - **背景**：`embedding` / `reranker` 的 `load()` 判断「要不要下载」原先只看 `config.json` 在不在。
   - ⚠️ **`config.json` 是第一个下的、权重是最后一个**（BGE-M3 2.2 GB / bge-reranker-base 1.1 GB，最容易中断）。一次被打断的首次下载会留下「config 在、权重不在」的目录 → 之后每次都跳过下载 → 在 transformers 深处报 `no file named model.safetensors, or pytorch_model.bin`，**与真因无关，且永远不会自愈**（只能手工删目录）。
   - **改法**（V1.37）：新增 `hf_mirror.is_complete(directory, weights)` —— **配置在**且**至少一个非空权重在**；两个适配层的 `load()` 与 `download()` 收尾校验**共用**它；`download_repo` 的跳过规则同样统一成「在且非空」（**0 字节一律当成「不在」**——镜像那条 `HEAD → /api/resolve-cache/` 的老路会留下 0 字节文件）。
   - ⚠️ **「存在」不等于「完整」**：判据必须覆盖**最容易失败的那一部分**，而不是最先到手的标记文件。
   - ⚠️ **模型目录（`EMBEDDING_MODEL_DIR`，默认 `data/models`）下的目录不是「配置齐了」就能用的**。遇到 `no file named model.safetensors` 这类报错，先去看那个目录**有没有权重**，不要先怀疑代码。
   - **怎么验**：`tests/test_hf_mirror.py`（11 项）——**已确认把修复还原后 9 项会红**。

32. **窄屏触控目标 ≥ 44×44 px**（`doc/前端界面说明.md` §4、§8.1）
   - **要求**：`< lg`（992px）的**可点击控件**——按钮、图标按钮、菜单项、导航链接——一律 ≥ 44×44 px（Apple HIG 与 WCAG 2.1 的触控目标建议）。lg 以上是桌面、鼠标精度足够，**保持 antd 默认的 32px**，否则整页按钮都会变笨重。
   - **落地**：`src/style.css` 里一条 `@media (max-width: 991px)`（断点与 `AppShell` 的 `useDrawer = !screens.lg` **同源**，别写成两套）。`.ant-btn` 加 `min-height: 44px`、`.ant-btn-icon-only` 加 `min-width: 44px`、`.ant-drawer-close` 44×44、`.ant-menu-item` 整行 44px。
   - ⚠️ **菜单项还要让行内链接铺满整行**（`.ant-menu-item .ant-menu-title-content > a { display:flex; height:44px; width:100% }`）——否则行是 44 高，可热区还是文字那一小块，**点行的空白处不会跳转**（实测「问答」只有 28×18）。
   - ⚠️ **只撑大盒子，不动图标与文字的字号**：`min-height` / `min-width` 会盖住 antd 写死的 `height` / `width`，**不需要 `!important`**。
   - **怎么验**：真实浏览器（`playwright-core` + 本机 Chrome，`ui-editor` 账号）逐页遍历问答 / 条款检索 / 文献管理 / 待审队列，窄屏另把抽屉打开量菜单项与关闭按钮，逐个 `getBoundingClientRect`。**375 / 768 应全部达标**；1280 有未达标项是**预期**。
